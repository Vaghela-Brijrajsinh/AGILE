package EXAM;

import org.openqa.selenium.By;
import org.openqa.selenium.Keys;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.WebElement;
import org.openqa.selenium.chrome.ChromeDriver;

public class pestrashop {

    public static void main(String[] args) throws Exception {

        WebDriver driver = new ChromeDriver();

        driver.manage().window().maximize();

        // 1. OPEN WEBSITE
        driver.get("https://demo.prestashop.com/#/en/front");

        Thread.sleep(15000);

        // 2. SWITCH TO IFRAME
        WebElement frame = driver.findElement(By.id("framelive"));
        driver.switchTo().frame(frame);

        Thread.sleep(3000);

        // 3. SEARCH PRODUCT
        driver.findElement(By.name("s"))
              .sendKeys("Hummingbird printed t-shirt", Keys.ENTER);

        Thread.sleep(5000);

        System.out.println("Product searched...");

        // 4. OPEN FIRST PRODUCT
        driver.findElement(By.cssSelector(".product-miniature a"))
              .click();

        Thread.sleep(4000);

        System.out.println("Product opened...");

        // 5. ADD TO CART
        Thread.sleep(3000);

        WebElement cartButton = driver.findElement(
            By.cssSelector("button[data-button-action='add-to-cart']")
        );

        cartButton.click();

        Thread.sleep(3000);

        System.out.println("Product added to cart...");

        // 6. OPEN CART
        driver.findElement(
            By.cssSelector(".modal a.btn-primary")
        ).click();

        Thread.sleep(3000);

        System.out.println("Cart opened...");

        // 7. CLOSE BROWSER
       // driver.quit();
    }
}
