package EXAM;

import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.chrome.ChromeDriver;
import org.openqa.selenium.chrome.ChromeOptions;



public class saucedemo {

	public static void main(String[] args) throws Exception
	{
		//REMOVE POPUP
		ChromeOptions options = new ChromeOptions();
		
		options.setExperimentalOption("prefs", java.util.Map.of(
					"credentails_enable_service","false",
					"prodile.password_manager_leak_detection",false
				
				));
		WebDriver driver = new ChromeDriver(options);
		
		driver.get("https://www.saucedemo.com");
		
		driver.manage().window().maximize();
		
		driver.findElement(By.id("user-name")).sendKeys("standard_user");
		
		driver.findElement(By.id("password")).sendKeys("secret_sauce");
		
		driver.findElement(By.id("login-button")).click();
		
		Thread.sleep(2000);
		
		driver.findElement(By.id("add-to-cart-sauce-labs-backpack")).click();
		
		driver.findElement(By.className("shopping_cart_link")).click();

		Thread.sleep(2000);
		
		driver.findElement(By.id("checkout")).click();
		
		driver.findElement(By.id("first-name")).sendKeys("first-name");
		
		driver.findElement(By.id("last-name")).sendKeys("last-name");
		
		driver.findElement(By.id("postal-code")).sendKeys("000000");
		
		Thread.sleep(1000);
		
		driver.findElement(By.id("continue")).click();
		

		driver.findElement(By.id("finish")).click();
		
		driver.findElement(By.id("generate-pdf-order")).click();
		
		
	}
}
